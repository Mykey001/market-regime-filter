input int FastMA = 10;
input int SlowMA = 20;

void OnTick()
{
    double fast[], slow[];
    CopyBuffer(handleFast, 0, 0, 3, fast);
    CopyBuffer(handleSlow, 0, 0, 3, slow);
    
    // Crossover detection
    bool BuySignal = (fast[1] > slow[1] && fast[2] <= slow[2]);
    bool SellSignal = (fast[1] < slow[1] && fast[2] >= slow[2]);
    
    if(BuySignal)
        OrderSend(...);  // Execute immediately
    
    if(SellSignal)
        OrderSend(...);  // Execute immediately
}

//|                                        MA_Crossover.mq5          |
//+------------------------------------------------------------------+

// CHANGE 1: Add this line
#include "MT5_RegimeFilter.mq5"

input int FastMA = 10;
input int SlowMA = 20;

void OnTick()
{
    double fast[], slow[];
    CopyBuffer(handleFast, 0, 0, 3, fast);
    CopyBuffer(handleSlow, 0, 0, 3, slow);
    
    // Crossover detection
    bool BuySignal = (fast[1] > slow[1] && fast[2] <= slow[2]);
    bool SellSignal = (fast[1] < slow[1] && fast[2] >= slow[2]);
    
    // CHANGE 2: Add && IsTradeAllowed()
    if(BuySignal && IsTradeAllowed("buy"))
        OrderSend(...);  // Only if regime allows
    
    if(SellSignal && IsTradeAllowed("sell"))
        OrderSend(...);  // Only if regime allows
}

    // Only trade with high confidence

    // Different strategy per regime
    switch(regime)
    {
        case 1:  // Normal/Calm → Mean Reversion
            if(RSI < 30 && IsTradeAllowed("buy"))
                BuyMeanReversion();
            if(RSI > 70 && IsTradeAllowed("sell"))
                SellMeanReversion();
            break;
            
        case 3:  // Bullish Trending → Trend Following
            if(PriceAboveMA && IsTradeAllowed("buy"))
                BuyTrend();
            break;

                SellTrend();
            break;
            
        case 4:  // Extreme Vol → Stay Out
        case 5:  // Crisis → Stay Out

            break;
    }
}
